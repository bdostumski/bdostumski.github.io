*1. Docker Build Basics*

*2. Multi-Stage Dockerfile Fundamentals*

#+BEGIN_SRC bash

# Example: Basic multi-stage Dockerfile
FROM maven:3.9-eclipse-temurin-17 AS build

WORKDIR /app

COPY pom.xml .
COPY src ./src

RUN mvn clean package -DskipTests

FROM eclipse-temurin:17-jre

WORKDIR /app

COPY --from=build /app/target/*.jar app.jar

ENTRYPOINT ["java", "-jar", "app.jar"]

# Build the multi-stage image
docker build -t java-app:1.0 .

# Run the final image
docker run -d --name java-app java-app:1.0

# Inspect the final image
docker image inspect java-app:1.0

# Display image size
docker images java-app

#+END_SRC

*3. Named Build Stages*

#+BEGIN_SRC bash

# Define a named build stage
FROM node:22 AS builder

WORKDIR /app

COPY package*.json ./
RUN npm ci

COPY . .
RUN npm run build

# Define a runtime stage
FROM nginx:alpine AS production

COPY --from=builder /app/dist /usr/share/nginx/html

# Build a specific stage
docker build --target builder -t myapp:builder .

# Build the final production stage
docker build --target production -t myapp:production .

# Display available build targets
docker build --call=targets .

# Build with a specific Dockerfile
docker build -f Dockerfile.multistage -t myapp:latest .

#+END_SRC

*4. Copy Artifacts Between Stages*

#+BEGIN_SRC bash

# Copy a single file from another stage
COPY --from=build /app/app.jar /app/app.jar

# Copy an entire directory
COPY --from=build /app/dist /usr/share/nginx/html

# Copy multiple artifacts
COPY --from=build /app/target/app.jar /app/app.jar
COPY --from=build /app/config /app/config

# Copy a specific executable
COPY --from=builder /app/bin/server /usr/local/bin/server

# Copy files from an external image
COPY --from=nginx:alpine /etc/nginx/nginx.conf /etc/nginx/nginx.conf

# Copy from a previous named stage
COPY --from=builder /app/build ./build

# Copy with specific ownership
COPY --from=builder --chown=10001:10001 \
  /app/app.jar /app/app.jar

# Copy with file permissions
COPY --from=builder --chmod=755 \
  /app/bin/server /usr/local/bin/server

# Inspect copied files inside the container
docker run --rm myapp:latest ls -la /app

#+END_SRC

*5. Multi-Stage Build with Java & Maven*

# Example Dockerfile
FROM maven:3.9-eclipse-temurin-17 AS build

# Copy dependency configuration first
COPY pom.xml .

# Download dependencies

# Copy application source

docker build --target build -t spring-app:build .

*7. Multi-Stage Build with Node.js*

#+BEGIN_SRC bash

# Example Dockerfile
FROM node:22-alpine AS dependencies

WORKDIR /app

COPY package*.json ./

RUN npm ci

FROM node:22-alpine AS build

WORKDIR /app

COPY --from=dependencies /app/node_modules ./node_modules

COPY . .

RUN npm run build

FROM node:22-alpine AS production-dependencies

WORKDIR /app

COPY package*.json ./

RUN npm ci --omit=dev

FROM node:22-alpine AS production

WORKDIR /app

ENV NODE_ENV=production

COPY --from=production-dependencies \
  /app/node_modules ./node_modules

COPY --from=build /app/dist ./dist

COPY package*.json ./

USER node

CMD ["node", "dist/main.js"]

# Build production image
docker build -t node-app:1.0 .

# Build only the development dependencies stage
docker build --target dependencies -t node-app:deps .

# Build application stage
docker build --target build -t node-app:build .

# Run application
docker run -d -p 3000:3000 node-app:1.0

#+END_SRC

*8. Multi-Stage Build with Go*

#+BEGIN_SRC bash

# Example Dockerfile
FROM golang:1.24-alpine AS build

WORKDIR /app

COPY go.mod go.sum ./

RUN go mod download

COPY . .

RUN CGO_ENABLED=0 GOOS=linux \
  go build -o server .

# Minimal runtime image
FROM scratch

COPY --from=build /app/server /server

ENTRYPOINT ["/server"]

# Build image
docker build -t go-app:1.0 .

# Build for Linux AMD64
docker build \
  --platform linux/amd64 \
  -t go-app:amd64 .

# Build for Linux ARM64
docker build \
  --platform linux/arm64 \
  -t go-app:arm64 .

# Run application
docker run -d --name go-app go-app:1.0

# Note:
# scratch contains no shell, package manager, or system libraries.
# The application must be compatible with the minimal runtime environment.

#+END_SRC

*9. Multi-Stage Build with Python*

#+BEGIN_SRC bash

# Example Dockerfile

  --wheel-dir /wheels \
  -r requirements.txt

WORKDIR /app

RUN pip install --no-cache-dir \
  --no-index \
  --find-links=/wheels \
  -r requirements.txt

# Build image

*11. BuildKit Cache Mounts*

#+BEGIN_SRC bash

# Example Dockerfile with Maven cache
# syntax=docker/dockerfile:1

FROM maven:3.9-eclipse-temurin-17 AS build

WORKDIR /app

COPY pom.xml .

RUN --mount=type=cache,target=/root/.m2 \
  mvn dependency:go-offline

COPY src ./src

RUN --mount=type=cache,target=/root/.m2 \
  mvn package -DskipTests

# BuildKit cache for npm

RUN --mount=type=cache,target=/root/.npm \
  npm ci

# BuildKit cache for pip

RUN --mount=type=cache,target=/root/.cache/pip \
  pip install -r requirements.txt

# Build with cache support
docker buildx build -t myapp:1.0 .

# Export build cache to registry
docker buildx build \
  --cache-to=type=registry,ref=username/myapp:buildcache,mode=max \
  -t username/myapp:1.0 \
  --push .

# Import build cache from registry
docker buildx build \
  --cache-from=type=registry,ref=username/myapp:buildcache \
  -t username/myapp:1.0 .

#+END_SRC

*12. Build Arguments & Environment Variables*

#+BEGIN_SRC bash

# Define build argument in Dockerfile
ARG APP_VERSION=1.0

# Use build argument
LABEL version=${APP_VERSION}

# Pass argument during build
docker build \
  --build-arg APP_VERSION=2.0 \
  -t myapp:2.0 .

# Multiple build arguments
docker build \
  --build-arg APP_VERSION=2.0 \
  --build-arg BUILD_ENV=production \
  -t myapp:production .

# Define environment variable
ENV NODE_ENV=production

# Display image environment variables
docker image inspect \
  --format='{{.Config.Env}}' myapp:1.0

# Pass runtime environment variable
docker run -e SPRING_PROFILES_ACTIVE=prod myapp:1.0

# Important:
# ARG is available during image building.
# ENV is available in the resulting image and at runtime.

#+END_SRC

*13. BuildKit Secrets & SSH Forwarding*

#+BEGIN_SRC bash

# Example: Mount a secret during build
# syntax=docker/dockerfile:1

FROM alpine AS build

RUN --mount=type=secret,id=mysecret \
  cat /run/secrets/mysecret

# Pass secret during build
docker build \
  --secret id=mysecret,src=./secret.txt \
  -t myapp:1.0 .

# Example: Use SSH forwarding

# syntax=docker/dockerfile:1

FROM alpine AS build

RUN apk add --no-cache git openssh-client

RUN --mount=type=ssh \
  git clone git@github.com:username/private-repo.git

# Build with SSH agent forwarding
docker build --ssh default -t myapp:1.0 .

# Use a specific SSH key
docker build \
  --ssh default=$HOME/.ssh/id_ed25519 \
  -t myapp:1.0 .

# Security recommendation:
# Never pass passwords, tokens, or private keys using ARG or ENV.
# Use BuildKit secret mounts for sensitive build-time information.

#+END_SRC

*19. Multi-Stage Build Security*

#+BEGIN_SRC bash

# Run the final image as a non-root user
USER 10001

# Copy artifact with specific ownership
COPY --from=build --chown=10001:10001 \
  /app/app.jar /app/app.jar

# Copy executable with restricted permissions
COPY --from=build --chmod=755 \
  /app/server /app/server

# Use a read-only root filesystem at runtime
docker run --read-only myapp:production

# Drop Linux capabilities
docker run --cap-drop ALL myapp:production

# Prevent privilege escalation
docker run \
  --security-opt no-new-privileges:true \
  myapp:production

# Scan final image using Docker Scout
docker scout quickview myapp:production

docker scout cves myapp:production

# Generate an SBOM using Docker Scout
docker scout sbom myapp:production

# Security checklist:
# Do not copy secrets into build stages unnecessarily.
# Use BuildKit secrets for private dependencies.
# Run applications as non-root users.
# Remove build tools from runtime images.
# Use trusted and version-pinned base images.
# Scan the final image.
# Generate and maintain an SBOM.
# Sign and verify production images.

#+END_SRC

8. Generate SBOM and provenance.

*25. Multi-Stage Build Best Practices*

1. Use named stages instead of numeric stage references.
2. Separate compilation from runtime execution.
3. Keep the final image as small as practical.
4. Copy only required artifacts between stages.
5. Use .dockerignore to reduce build context.
6. Optimize Dockerfile instruction ordering.
7. Cache dependencies before copying frequently changing source code.
8. Use BuildKit cache mounts for package managers.
9. Use BuildKit secret mounts for credentials.
10. Prefer non-root runtime users.
11. Check native library compatibility between stages.
12. Test the final image, not only the builder stage.
13. Pin base image versions and manage updates.
14. Scan and generate SBOMs for release images.
15. Use multi-platform builds when deploying across architectures.

* Beginner

** Q1: What is a Docker multi-stage build?
A Docker multi-stage build uses multiple ~FROM~ instructions in a single Dockerfile to build artifacts in one stage and copy only the needed result into a smaller final image.

** Q2: Why do we need multi-stage builds?
They reduce image size, improve security, and keep build tooling out of the final runtime image.

** Q3: What is a Dockerfile?
A Dockerfile is a text file that defines how an image is built.

** Q4: What is a build stage?
A build stage is a section of the Dockerfile starting with a ~FROM~ instruction.

** Q5: What is a final stage?
The final stage is the last stage in the Dockerfile and is the image that gets produced as the final artifact.

** Q6: What is the purpose of a build stage?
A build stage can compile code, install dependencies, or generate artifacts.

** Q7: What is a runtime image?
A runtime image is the final image used to run the application in production or development.

** Q8: What is the main benefit of multi-stage builds?
They allow builders and compilers to remain in earlier stages while the final image stays lean.

** Q9: What does ~COPY --from=...~ do?
It copies files from one stage into another stage in a multi-stage Dockerfile.

** Q10: What is image layering?
Image layering is how Docker composes read-only filesystem layers to build an image.

** Q11: Why is the final image usually smaller with multi-stage builds?
Because only necessary files are copied from the build stage into the final image.

** Q12: What is a build dependency?
A build dependency is a tool or package required only while compiling or packaging the application.

** Q13: What is a runtime dependency?
A runtime dependency is a library or package needed when the application actually runs.

** Q14: What is a compiled artifact?
A compiled artifact is the output of the build step, such as a binary, JAR, WAR, or static files.

** Q15: Why do compiled artifacts matter?
They are usually the only things needed in the final runtime image.

** Q16: What is a builder image?
A builder image contains the toolchain required to compile the project.

** Q17: What is a distroless image?
A distroless image contains only the runtime dependencies and not a package manager or shell.

** Q18: Why do developers like distroless images?
They reduce attack surface and image size.

** Q19: What is a base image?
A base image is the starting point for a Docker image build.

** Q20: What is a final image after multi-stage build?
It is the image produced by the last stage and served to runtime workloads.

** Q21: Why remove build tools from runtime images?
To reduce attack surface and avoid unnecessary package installation.

** Q22: What is the typical pattern in a multi-stage Dockerfile?
Build in one stage, then copy results into a smaller runtime stage.

** Q23: What is a common cross-compilation scenario?
Using a build image with the full toolchain to compile for a target environment and then shipping a minimal runtime image.

** Q24: What is a Docker stage name?
A Docker stage can be assigned a name using ~AS stage_name~ to make copying easier.

** Q25: What is the role of ~AS~ in Dockerfile?
~AS~ assigns a name to a stage so it can be referenced later with ~COPY --from=stage_name~.

** Q26: What is a stage dependency?
A stage dependency is when one stage depends on artifacts produced by another stage.

** Q27: What is a compile-only dependency?
A dependency required for compilation but not needed at runtime.

** Q28: What is a package manager in build stages?
A package manager such as apt, apk, yum, npm, or Maven is often used during compilation.

** Q29: What is a package manager in runtime stage?
