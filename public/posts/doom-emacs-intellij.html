" ============================================================
" IdeaVim configuration — Doom Emacs workflow for IntelliJ IDEA
" ============================================================
"
" REQUIRED IDE PLUGINS (Settings > Plugins > Marketplace):
"   * IdeaVim                     <- Vim like keybindings
"   * Which-Key                   <- The whole point of a Doom layout
"   * AceJump                     <- SPC j jump motions
"   * IdeaVim-EasyMotion          <- Optional, needs AceJump
"
" If a plugin is missing, IdeaVim logs "unknown option" on startup.
"
" ------------------------------------------------------------
" LEADER
" ------------------------------------------------------------
" NOTE: localleader is "," (was SPC — identical leaders make every <localleader> map dead weight).

let mapleader=" "
let maplocalleader=","


" ------------------------------------------------------------
" PLUGINS
" ------------------------------------------------------------

set surround                " ys / cs / ds
set commentary              " gcc, gc{motion}
set ReplaceWithRegister     " gr{motion}, grr
set exchange                " cx{motion}, cxx
set argtextobj              " ia / aa
set textobj-entire          " ie / ae
set textobj-indent          " ii / ai
set matchit                 " % over tags/keywords
set vim-paragraph-motion    " { } over whitespace-only lines
set highlightedyank
set multiple-cursors        " <A-n> / <A-x> / <A-p>
set sneak                   " s{char}{char} / S{char}{char}
set NERDTree                " :NERDTree drives the Project view
set which-key
let g:EasyMotion_do_mapping = 0   " no default <leader><leader> maps (clashes with SPC SPC)
set easymotion
let g:argtextobj_pairs="[:],(:),<:>"
let g:highlightedyank_highlight_duration = "200"


" ------------------------------------------------------------
" GENERAL OPTIONS
" ------------------------------------------------------------

set number
set relativenumber

set ignorecase
set smartcase
set hlsearch
set incsearch

set scrolloff=8
set sidescrolloff=8

set visualbell
set showmode
set showcmd

" which-key needs a long/absent timeout or the popup races you
set notimeout
set timeoutlen=5000

set clipboard=unnamedplus,unnamed,ideaput

" ============================================================
" EDITING QUALITY OF LIFE
" ============================================================

inoremap jk <Esc>
inoremap jj <Esc>

" Q was mapped to gq, which IdeaVim does not implement. Replay @q instead.
nnoremap Q @q

" Yank to end of line, like D and C
nnoremap Y y$

" Keep the cursor parked while scrolling / searching / joining
nnoremap <C-d> <C-d>zz
nnoremap <C-u> <C-u>zz
nnoremap n nzzzv
nnoremap N Nzzzv
nnoremap J mzJ`z

" x and single-char deletes must not clobber the yank register
nnoremap x "_x

" Visual mode
vnoremap < <gv
vnoremap > >gv
xnoremap p "_dP
vnoremap J :m '>+1<CR>gv=gv
vnoremap K :m '<-2<CR>gv=gv

" Blank lines around the cursor (vim-unimpaired style)
nnoremap [<Space> O<Esc>j
nnoremap ]<Space> o<Esc>k

" DOOM-STYLE LEADER MAP
"
" SPC SPC  M-x / search everywhere      SPC h   help
" SPC .    find file                    SPC i   insert
" SPC ,    switch buffer                SPC j   jump
" SPC `    last editor                  SPC m   major mode (Java/Maven)
" SPC /    search project               SPC n   bookmarks / notes
" SPC :    find action                  SPC o   open (tool windows)
" SPC b    buffer                       SPC p   project
" SPC c    code + refactor              SPC q   quit / session
" SPC d    debug                        SPC r   run
" SPC e    errors                       SPC s   search
" SPC f    file                         SPC t   test   (Doom uses toggle)
" SPC g    git                          SPC T   toggle
